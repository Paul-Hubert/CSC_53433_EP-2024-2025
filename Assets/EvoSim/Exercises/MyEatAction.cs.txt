/// <summary>Exercise 4: go to the nearest visible food and eat it (ACT-04, ACT-10). Reference: <see cref="EatAction"/>.</summary>
/// <remarks>
/// <para>Write <see cref="Act"/>. The settings, the prompt line and <see cref="IsRelevant"/> are given, as in the reference.</para>
/// <para>The reference also eats eggs when <c>target</c> is "egg:&lt;species&gt;" (22 §15): in <c>Initialize</c> it finds that
/// species once (<c>World.FindSpeciesByName</c>); in Act it walks to the nearest egg in sight (<c>c.NearestEgg(a, species)</c>)
/// and eats it on arrival (<c>c.OnArrival(Interaction.EatEgg(egg.Value.Entity))</c>), or searches when there is none.</para>
/// </remarks>
[ExerciseOf(typeof(EatAction))]
public class MyEatAction : AnimalAction
{
    [SerializeField, Tooltip("A layer name; empty = every layer the diet grazes; \"egg:<species>\" = eggs (22 §15).")]
    string target = "";
    [SerializeField, Tooltip("The sense whose token says whether food is in sight (CTRL-10).")]
    string relevantSense = "Food";

    /// <summary>The action's line in the prompt (08 §7), used when the Inspector's description is empty.</summary>
    protected override string DefaultDescription => "go to the nearest visible food and eat it";

    /// <summary>
    /// Find the nearest food in sight (ActContext), walk to it, graze on arrival; nothing in sight: search. The reference
    /// asks <c>c.NearestResource(a, target)</c> (within vision, only layers the diet grazes); with none it calls
    /// <c>c.Search()</c> (ACT-04). Already within <c>c.GrazeReach</c> of the item it stays, else it walks to the item's
    /// position without stopping short (<c>c.WalkTo(position, 0f)</c>, MOVE-04). Either way it asks to graze on arrival,
    /// <c>c.OnArrival(Interaction.Graze(food.Value))</c>, which happens on the very tick it arrives (ACT-10).
    /// </summary>
    /// <inheritdoc/>
    public override void Act(Animal a, ActContext c) => throw new NotImplementedException(ExerciseOfAttribute.Todo("MyEatAction", "Act"));

    /// <summary>True when the sense named by <c>relevantSense</c> sees food: its token is neither missing nor "none" (CTRL-10).</summary>
    /// <inheritdoc/>
    public override bool IsRelevant(ObservationView o)
    {
        var token = o.Token(relevantSense);
        return token != null && token != "none";
    }
}
